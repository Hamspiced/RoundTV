cmake_minimum_required(VERSION 3.16)
project(gif20_check C CXX)
if(NOT LVGL_DIR)
  message(FATAL_ERROR "Set LVGL_DIR to the LVGL 8.3.11 component directory")
endif()
set(LV_CONF_PATH "${CMAKE_CURRENT_SOURCE_DIR}/lv_conf.h" CACHE STRING "" FORCE)
add_subdirectory("${LVGL_DIR}" lvgl-build)
add_executable(gif20_check main.cpp ../../src/gif_player.cpp)
target_include_directories(gif20_check PRIVATE ../../src)
target_link_libraries(gif20_check lvgl m)
target_compile_features(gif20_check PRIVATE cxx_std_17)
